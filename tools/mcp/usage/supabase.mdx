This Agent uses the [Supabase MCP server](https://github.com/supabase/mcp) to create projects, database schemas, and edge functions.

```python
"""Supabase MCP Agent - Showcase Supabase MCP Capabilities

This example demonstrates how to use the Supabase MCP server to create projects, database schemas, edge functions, and more.

Setup:
1. Install Python dependencies: `uv pip install "agno[mcp]" openai`
2. Create a Supabase Access Token: https://supabase.com/dashboard/account/tokens and set it as the SUPABASE_ACCESS_TOKEN environment variable.
3. Set the OPENAI_API_KEY environment variable.
"""

import asyncio
import os
from textwrap import dedent

from agno.agent import Agent
from agno.models.openai import OpenAIResponses
from agno.tools.mcp import MCPTools
from agno.tools.reasoning import ReasoningTools
from agno.utils.log import log_error, log_exception, log_info


async def run_agent(task: str) -> None:
    token = os.getenv("SUPABASE_ACCESS_TOKEN")
    if not token:
        log_error("SUPABASE_ACCESS_TOKEN environment variable not set.")
        return

    npx_cmd = "npx.cmd" if os.name == "nt" else "npx"
    organization_id = input("Supabase organization ID: ").strip()
    if not organization_id:
        log_error("A Supabase organization ID is required.")
        return

    try:
        async with MCPTools(
            f"{npx_cmd} -y @supabase/mcp-server-supabase@latest --access-token={token}",
            include_tools=[
                "list_organizations",
                "get_cost",
                "confirm_cost",
                "create_project",
                "get_project",
                "apply_migration",
                "list_tables",
                "list_extensions",
                "deploy_edge_function",
                "get_project_url",
                "get_publishable_keys",
            ],
            requires_confirmation_tools=[
                "confirm_cost",
                "create_project",
                "apply_migration",
                "deploy_edge_function",
            ],
        ) as mcp:
            instructions = dedent(f"""
                You are an expert Supabase MCP architect. Given the project description:
                {task}

                Use organization ID `{organization_id}`. Never infer or substitute another organization ID.

                Perform the following steps:
                1. Plan the entire database schema based on the project description.
                2. Call `list_organizations` and verify that the provided organization ID is present.
                3. Use `get_cost(type='project', organization_id='{organization_id}')` to estimate project creation cost.
                4. Call `confirm_cost` with the cost details returned by `get_cost`.
                5. Create a new Supabase project with `create_project`.
                6. Poll project status with `get_project` until the status is `ACTIVE_HEALTHY`.
                7. Analyze the project requirements and propose a complete, normalized SQL schema (tables, columns, data types, indexes, constraints, triggers, and functions) as DDL statements.
                8. Apply the schema using `apply_migration`, naming the migration `initial_schema`.
                9. Validate the deployed schema via `list_tables` and `list_extensions`.
                10. Deploy a simple health-check edge function with `deploy_edge_function`.
                11. Retrieve and print the project URL (`get_project_url`) and publishable keys (`get_publishable_keys`).
            """)
            agent = Agent(
                model=OpenAIResponses(id="gpt-5.2"),
                instructions=instructions,
                tools=[mcp, ReasoningTools(add_instructions=True)],
                markdown=True,
            )

            log_info(f"Running Supabase project agent for: {task}")
            run_response = await agent.arun(
                input=f"Organization ID: {organization_id}\n\nProject: {task}",
                show_full_reasoning=True,
            )
            while run_response.is_paused:
                for requirement in run_response.active_requirements:
                    if not requirement.needs_confirmation:
                        continue
                    execution = requirement.tool_execution
                    print(f"\nApprove {execution.tool_name}({execution.tool_args})?")
                    if input("[y/N] ").strip().lower() == "y":
                        requirement.confirm()
                    else:
                        requirement.reject("User rejected this operation.")

                run_response = await agent.acontinue_run(
                    run_id=run_response.run_id,
                    requirements=run_response.requirements,
                )

            print(run_response.content)
    except Exception as e:
        log_exception(f"Unexpected error: {e}")


if __name__ == "__main__":
    demo_description = (
        "Develop a cloud-based SaaS platform with AI-powered task suggestions, calendar syncing, predictive prioritization, "
        "team collaboration, and project analytics."
    )
    asyncio.run(run_agent(demo_description))
```
